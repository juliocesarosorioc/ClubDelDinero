LÓGICA DE INTERFAZ (UI RENDERING) // ========================================== const UI = { renderSaldos: () => { const panel = document.getElementById('panelSaldos'); // Mapeamos y usamos join('') para evitar repintados múltiples del DOM (Mejora de Rendimiento) panel.innerHTML = Object.keys(AppState.usuarios).map(u => `
${u}: USD
`).join(''); // Delegación de eventos (Más ligero que onClick en el HTML) document.querySelectorAll('.input-saldo').forEach(input => { input.addEventListener('change', (e) => { AppState.usuarios[e.target.dataset.user] = parseFloat(e.target.value); }); }); }, generarPizarra: () => { const cant = parseInt(document.getElementById('numPuestos').value) || 4; const cont = document.getElementById('puestosContainer'); let html = ''; for(let i=1; i<=cant; i++) { let valPredeterminado = i <= 4 ? ${i}º
`; } cont.innerHTML = html; } }; // ========================================== // 3. LÓGICA DE NEGOCIO (CORE BUSINESS) // ========================================== const LogicaHipica = { procesarPlano: () => { const texto = document.getElementById('planoTexto').value; const lineas = texto.split('\n').filter(l => l.trim() !== ''); // Filtra líneas vacías rápido let saldosTemp = { ...AppState.usuarios }; let sobregiros = []; let historialPorUsuario = {}; AppState.jugadasAprobadas = []; // Arrays para recolectar HTML y evitar innerHTML en bucle let filasEvaluacionHTML = []; let planoGrupoTxt = "📋 *PLANO OFICIAL DE JUGADAS*\n\n"; lineas.forEach((linea, index) => { const partes = linea.trim().split(/\s+/); if (partes.length < 5) return; const juega = partes[0]; const montoSol = parseFloat(partes[partes.length - 1]); const da = partes[partes.length - 2]; const caballo = partes[partes.length - 3]; const jugada = partes.slice(1, partes.length - 3).join(' '); if(!historialPorUsuario[juega]) historialPorUsuario[juega] = []; const sJ = saldosTemp[juega] || 0; const sD = saldosTemp[da] || 0; let rD = jugada.toUpperCase() === '10A7' ? montoSol * 0.7 : montoSol; // Cálculo de factor limitante const factorLim = Math.min((sJ / montoSol) || 0, (sD / rD) || 0, 1.0); const montoAuth = Math.floor(montoSol * factorLim); saldosTemp[juega] -= (montoAuth * (montoSol/montoSol)); saldosTemp[da] -= (montoAuth * (rD/montoSol)); // Determinar estado visual const estado = montoAuth === 0 ? { emoji: "❌", badge: "err", text: "RECHAZADO" } : montoAuth < montoSol ? { emoji: "⚠️", badge: "warn", text: "LIMITADO" } : { emoji: "✅", badge: "ok", text: "APROBADO" }; planoGrupoTxt += `${estado.emoji} ${juega} ${jugada} ${caballo} ${da} ${montoAuth}${montoAuth < montoSol ? ' (Ajuste)' : ''}\n`; if (montoAuth < montoSol) sobregiros.push(juega); historialPorUsuario[juega].push({ da, jugada, caballo, pedido: montoSol, autorizado: montoAuth, saldoRst: saldosTemp[juega] }); if (montoAuth > 0) AppState.jugadasAprobadas.push({ juega, jugada, caballo, da, monto: montoAuth }); filasEvaluacionHTML.push(` #${index + 1} ${juega} vs ${da} ${jugada} (#${caballo}) $${montoSol} $${montoAuth} $${saldosTemp[juega].toFixed(2)} ${estado.text} `); }); // Inserción masiva única (Super rápido) document.getElementById('tablaEvaluacion').innerHTML = filasEvaluacionHTML.join(''); planoGrupoTxt += "\n-----------------------------------\n📌 *Jugadas procesadas y garantizadas por la plataforma.*"; document.getElementById('outPlanoGrupo').innerText = planoGrupoTxt; // Lógica Privada LogicaHipica.mostrarMensajesPrivados(sobregiros, historialPorUsuario); }, mostrarMensajesPrivados: (sobregiros, historialPorUsuario) => { const secPriv = document.getElementById('seccionPrivada'); if (sobregiros.length === 0) { secPriv.style.display = 'none'; return; } const userAfectado = sobregiros[0]; // Tomamos el primer afectado para el ejemplo let msgPriv = `⚠️ *CONTROL DE SALDO - NOTIFICACIÓN*\n👤 *Usuario:* ${userAfectado}\n💰 *Saldo Inicial:* $${AppState.usuarios[userAfectado]} USD\n-------------------------------------------\n`; historialPorUsuario[userAfectado].forEach((j, i) => { let st = j.autorizado === j.pedido ? "✅ APROBADA" : (j.autorizado > 0 ? `⚠️ LIMITADA A $${j.autorizado}` : "❌ RECHAZADA"); msgPriv += `${i+1}. VS *${j.da}* (${j.jugada} #${j.caballo}) ➔ Pedido: $${j.pedido} ➔ ${st}\n`; }); document.getElementById('outMensajePrivado').innerText = msgPriv + `\n📌 Por favor recargar saldo para ampliar tus límites.`; secPriv.style.display = 'block'; } }; // ========================================== // 4. CONEXIÓN API (NODE.JS) - SEGURIDAD // ========================================== async function registrarTransaccionBackend() { const payload = { usuario: document.getElementById('transUsuario').value.trim().toUpperCase(), tipo: document.getElementById('transTipo').value, banco: document.getElementById('transBanco').value.trim().toUpperCase(), monto: parseFloat(document.getElementById('transMonto').value), referencia: document.getElementById('transRef').value.trim() }; if (!payload.usuario || !payload.banco || isNaN(payload.monto) || payload.monto <= 0 || !payload.referencia) { return alert("⚠️ Por favor completa todos los campos correctamente."); } // Deshabilitar botón temporalmente para evitar doble clic const btn = document.querySelector('.btn-green'); btn.innerText = "Registrando..."; btn.disabled = true; try { // Aquí llamamos a nuestro propio Backend Node.js, NO directamente a Supabase const response = await fetch(`${AppState.API_URL}/transacciones`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); const data = await response.json(); if (!response.ok) throw new Error(data.error || 'Error desconocido'); alert(`✅ ¡Transacción registrada con éxito!\nReferencia: ${payload.referencia}`); document.getElementById('transMonto').value = ''; document.getElementById('transRef').value = ''; } catch (error) { alert(`❌ Error: ${error.message}`); } finally { btn.innerText = "Registrar Operación"; btn.disabled = false; } } // ========================================== // INICIALIZACIÓN // ========================================== window.onload = () => { UI.renderSaldos(); UI.generarPizarra(); // Agregar listeners a los botones existentes document.querySelector('.btn-green').onclick = registrarTransaccionBackend; // (Asocia las demás funciones a los botones correspondientes) };
